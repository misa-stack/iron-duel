cmake_minimum_required(VERSION 3.16)
project(OcelovyDuel VERSION 1.0.0 LANGUAGES CXX)
option(DUEL_BUILD_GUI "Build the SDL2 graphical frontend" ON)
option(DUEL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(DUEL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(duel_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(duel_options INTERFACE -Wall -Wextra -Wpedantic -Wshadow)
    if(DUEL_WARNINGS_AS_ERRORS)
        target_compile_options(duel_options INTERFACE -Werror)
    endif()
    if(DUEL_SANITIZERS)
        target_compile_options(duel_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(duel_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()
add_library(duel_core src/terrain.cpp src/simulation.cpp src/ai.cpp src/replay.cpp)
target_include_directories(duel_core PUBLIC include)
target_link_libraries(duel_core PUBLIC duel_options)
add_executable(ocelovyduel src/main.cpp)
target_link_libraries(ocelovyduel PRIVATE duel_core)
if(DUEL_BUILD_GUI)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET "sdl2>=2.0.18")
    find_package(PNG REQUIRED)
    target_sources(ocelovyduel PRIVATE src/frontend.cpp src/renderer.cpp)
    target_link_libraries(ocelovyduel PRIVATE PkgConfig::SDL2 PNG::PNG)
    target_compile_definitions(ocelovyduel PRIVATE DUEL_WITH_SDL DUEL_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    configure_file(cislapismenamalaivelka.png cislapismenamalaivelka.png COPYONLY)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(duel_tests tests/core_tests.cpp)
    target_link_libraries(duel_tests PRIVATE duel_core)
    add_test(NAME core COMMAND duel_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:ocelovyduel>)
        set_tests_properties(cli PROPERTIES TIMEOUT 120)
        if(DUEL_BUILD_GUI)
            add_test(NAME gui COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_tests.py $<TARGET_FILE:ocelovyduel>)
            set_tests_properties(gui PROPERTIES TIMEOUT 120)
        endif()
    endif()
endif()
add_executable(terrain_bench benchmarks/terrain_bench.cpp)
target_link_libraries(terrain_bench PRIVATE duel_core)
install(TARGETS ocelovyduel RUNTIME DESTINATION .)
install(FILES cislapismenamalaivelka.png README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
set(CPACK_GENERATOR "TGZ")
set(CPACK_PACKAGE_FILE_NAME "ocelovyduel-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
